cmake_minimum_required(VERSION 3.21)
project(SummitExternalNative LANGUAGES CXX)
foreach(source_dir CABANA_SOURCE ARBORX_SOURCE BENCHMARK_SOURCE BOOST_SOURCE)
  file(TO_CMAKE_PATH "${${source_dir}}" ${source_dir})
endforeach()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Kokkos 4.5 REQUIRED CONFIG)
set(Cabana_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(Cabana_ENABLE_PERFORMANCE_TESTING OFF CACHE BOOL "" FORCE)
set(Cabana_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(Cabana_ENABLE_GRID OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_MPI ON)
set(CMAKE_DISABLE_FIND_PACKAGE_ArborX ON)
add_subdirectory("${CABANA_SOURCE}" cabana)
set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
add_subdirectory("${BENCHMARK_SOURCE}" benchmark)
# Same eleven translation units and static definitions as Boost 1.87's native
# program_options CMake target. The official archive provides all headers.
set(boost_names cmdline config_file convert options_description parsers positional_options split utf8_codecvt_facet value_semantic variables_map winmain)
foreach(name IN LISTS boost_names)
  list(APPEND boost_sources "${BOOST_SOURCE}/libs/program_options/src/${name}.cpp")
endforeach()
add_library(boost_program_options STATIC ${boost_sources})
target_include_directories(boost_program_options PUBLIC "${BOOST_SOURCE}")
target_compile_definitions(boost_program_options PUBLIC BOOST_PROGRAM_OPTIONS_NO_LIB BOOST_PROGRAM_OPTIONS_STATIC_LINK PRIVATE BOOST_PROGRAM_OPTIONS_SOURCE)
set(ARBORX_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(ARBORX_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ARBORX_ENABLE_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ARBORX_SOURCE}" arborx)
# Upstream skips this target on Windows due to a historical installed-benchmark
# link issue. Link the exact driver with our same-toolchain framework/dependency.
add_executable(ArborXNative "${ARBORX_SOURCE}/benchmarks/bvh_driver/bvh_driver.cpp")
target_link_libraries(ArborXNative PRIVATE ArborX::ArborX benchmark::benchmark boost_program_options)
target_include_directories(ArborXNative PRIVATE "${ARBORX_SOURCE}/benchmarks/utils" "${ARBORX_SOURCE}/test")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/CabanaReplay.cpp")
  add_executable(CabanaReplay CabanaReplay.cpp)
  target_link_libraries(CabanaReplay PRIVATE Cabana::Core)
endif()
if(CABANA_CAPTURE_SOURCE)
  add_executable(CabanaCapture "${CABANA_CAPTURE_SOURCE}")
  target_link_libraries(CabanaCapture PRIVATE Cabana::Core)
  target_include_directories(CabanaCapture PRIVATE "${CABANA_SOURCE}/benchmark" "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ArborXReplay.cpp")
  add_executable(ArborXReplay ArborXReplay.cpp)
  target_link_libraries(ArborXReplay PRIVATE ArborX::ArborX benchmark::benchmark boost_program_options)
  target_include_directories(ArborXReplay PRIVATE "${ARBORX_SOURCE}/benchmarks/bvh_driver" "${ARBORX_SOURCE}/benchmarks/utils" "${ARBORX_SOURCE}/test")
endif()
