cmake_minimum_required(VERSION 3.22)
project(SummitWholeTaskMD LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Kokkos 4.5 REQUIRED CONFIG)
set(MD_BACKEND "serial" CACHE STRING "Explicit execution space: serial, openmp, cuda")
set_property(CACHE MD_BACKEND PROPERTY STRINGS serial openmp cuda)
if(NOT MD_BACKEND MATCHES "^(serial|openmp|cuda)$")
  message(FATAL_ERROR "Unsupported MD_BACKEND")
endif()
string(TOUPPER "${MD_BACKEND}" MD_BACKEND_UPPER)
if(NOT Kokkos_ENABLE_${MD_BACKEND_UPPER})
  message(FATAL_ERROR "The selected execution space is absent from this Kokkos build")
endif()
set(ARBORX_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(ARBORX_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ARBORX_ENABLE_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(ARBORX_ENABLE_MPI OFF CACHE BOOL "" FORCE)
add_subdirectory("${ARBORX_SOURCE}" arborx)
add_executable(SummitMolecularNative main.cpp)
target_link_libraries(SummitMolecularNative PRIVATE ArborX::ArborX)
target_include_directories(SummitMolecularNative PRIVATE "${GENERATED_SOURCE}")
target_compile_definitions(SummitMolecularNative PRIVATE SUMMIT_MD_${MD_BACKEND_UPPER}=1)
add_executable(UpstreamMolecularOriginal "${ARBORX_SOURCE}/examples/molecular_dynamics/example_molecular_dynamics.cpp")
target_link_libraries(UpstreamMolecularOriginal PRIVATE ArborX::ArborX)
foreach(target SummitMolecularNative UpstreamMolecularOriginal)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /fp:strict)
  elseif(MD_BACKEND STREQUAL "cuda")
    # nvcc_wrapper compiles CXX sources. No fast math or FMA contraction.
    target_compile_options(${target} PRIVATE --fmad=false -Xcompiler=-ffp-contract=off)
  else()
    target_compile_options(${target} PRIVATE -fno-fast-math -ffp-contract=off)
  endif()
endforeach()
